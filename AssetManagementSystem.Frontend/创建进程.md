封装axios  后端是(http)
request.ts
逻辑：
1.创建一个 Axios 实例(后端是 HttpClient )：
    设置 baseURL（比如 /api，方便解决跨域问题）和 timeout。
2.请求拦截器  request （后端是 HttpClient.DefaultRequestHeaders ）：
    在每次发送请求前，从 localStorage 里取出 Token，塞进 Header 里（格式：Authorization: Bearer token）。
3.响应拦截器 response （后端是 HttpResponseMessage ）：
    统一处理后端返回的 ApiResult。如果 code是成功的，直接返回 data；如果是 401（未登录/Token过期），强制跳转到登录页。

//默认请求拦截器写法
    // 相当于自定义 DelegatingHandler 的 SendAsync 方法（发送前）
request.interceptors.request.use(
  (config) => {
    // C# 类比：在发送请求前，给 HttpClient 添加 Authorization Header
    // httpClient.DefaultRequestHeaders.Authorization = 
    //   new AuthenticationHeaderValue("Bearer", token);
    
    const token = localStorage.getItem('token');  // 相当于 session/缓存
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => Promise.reject(error)
);

//默认响应拦截器写法
request.interceptors.response.use(
  (response) => {
    // response 相当于 HttpResponseMessage
    // response.data 就是 ApiResult<T>
    
    const { code, data, message } = response.data;
    
    if (code === 200) {
      // ✅ 成功：只返回 data，相当于 C# 的 return result.Data
      return data;
    } else if (code === 401) {
      // ❌ 未授权：跳转到登录页
      localStorage.removeItem('token');
      window.location.href = '/login';
      return Promise.reject(new Error('未登录'));
    } else {
      // ❌ 业务错误：弹出提示
      // 相当于 C# 的 throw new BusinessException(message)
      ElMessage.error(message);
      return Promise.reject(new Error(message));
    }
  },
  (error) => {
    return Promise.reject(error);
  }
);

Promise 的核心概念 
1. Promise = 前端的 Task<T>                       
   不是工具方法，是异步编程模式                 
                                               
 2. Promise 是 JS 语言特性，前后端通用           
   Node.js 后端天天用 Promise                   
                                                
3. 你封装的 request 返回的就是 Promise          
   const data = await request.get('/user');     
  这里的 await 就是等待 Promise 完成           
                                                
4. 错误处理用 try-catch 还是 .catch() 都行      
   try { await request.get() } catch (e) {}  

const promise = new Promise((resolve, reject) => {
    // 状态1: pending（进行中）
    setTimeout(() => {
        // 状态2: fulfilled（成功）
        resolve('成功数据');
        // 状态3: rejected（失败）
        reject('错误信息');
    }, 1000);
});
